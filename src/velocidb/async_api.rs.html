1//! Native async API, modeled on Turso's Rust binding.
2//!
3//! ```rust,no_run
4//! use velocidb::async_api::Builder;
5//!
6//! # async fn example() -> velocidb::types::Result<()> {
7//! let db = Builder::new_local("app.db").build().await?;
8//! let conn = db.connect()?;
9//!
10//! conn.execute("CREATE TABLE users (id INTEGER PRIMARY KEY, name TEXT)").await?;
11//! conn.execute("INSERT INTO users VALUES (1, 'Alice')").await?;
12//!
13//! let rows = conn.query("SELECT * FROM users").await?;
14//! println!("{} row(s)", rows.rows.len());
15//! # Ok(())
16//! # }
17//! ```
18//!
19//! The engine itself is synchronous; this layer offloads every call onto the
20//! tokio blocking thread pool so async tasks never stall the reactor. Because
21//! readers run concurrently in the engine, multiple in-flight `query` futures
22//! execute in parallel across blocking threads.
23

30/// Builder for opening a database asynchronously (Turso-style entry point).
31pub struct Builder {
32    path: PathBuf,
33}
34
35impl Builder {
36    /// Creates a builder for a local database file.
37    pub fn new_local<P: Into<PathBuf>>(path: P) -> Self {
38        Self { path: path.into() }
39    }
40
41    /// Opens (or creates) the database without blocking the async runtime.
42    pub async fn build(self) -> Result<AsyncDatabase> {
43        let path = self.path;
44        let db = run_blocking(move || Database::open(path)).await?;
45        Ok(AsyncDatabase { inner: db })
46    }
47}

49/// An asynchronously opened database. Cheap to clone; hand out one
50/// [`AsyncConnection`] per task with [`AsyncDatabase::connect`].
51#[derive(Clone)]
52pub struct AsyncDatabase {
53    inner: Arc<Database>,
54}
55
56impl AsyncDatabase {
57    /// Creates a connection to this database.
58    pub fn connect(&self) -> Result<AsyncConnection> {
59        Ok(AsyncConnection {
60            inner: Arc::clone(&self.inner),
61        })
62    }
63}

65/// A connection with async statement execution.
66///
67/// Connections are cheap clones of the shared database handle. Writes are
68/// serialized by the engine; reads run concurrently.
69#[derive(Clone)]
70pub struct AsyncConnection {
71    inner: Arc<Database>,
72}
73
74impl AsyncConnection {
75    /// Executes a statement that returns no rows (DDL / DML).
76    pub async fn execute(&self, sql: &str) -> Result<()> {
77        let db = Arc::clone(&self.inner);
78        let sql = sql.to_string();
79        run_blocking(move || db.execute(&sql)).await
80    }
81
82    /// Executes a query and returns the result rows.
83    pub async fn query(&self, sql: &str) -> Result<QueryResult> {
84        let db = Arc::clone(&self.inner);
85        let sql = sql.to_string();
86        run_blocking(move || db.query(&sql)).await
87    }
88
89    /// Exact K-nearest-neighbour vector search (see [`Database::vector_search`]).
90    pub async fn vector_search(
91        &self,
92        table: &str,
93        column: &str,
94        query: &[f32],
95        k: usize,
96        metric: DistanceMetric,
97    ) -> Result<Vec<(f64, Row)>> {
98        let db = Arc::clone(&self.inner);
99        let table = table.to_string();
100        let column = column.to_string();
101        let query = query.to_vec();
102        run_blocking(move || db.vector_search(&table, &column, &query, k, metric)).await
103    }
104
105    /// Begins an explicit transaction.
106    pub async fn begin(&self) -> Result<()> {
107        let db = Arc::clone(&self.inner);
108        run_blocking(move || db.begin()).await
109    }
110
111    /// Commits the current explicit transaction.
112    pub async fn commit(&self) -> Result<()> {
113        let db = Arc::clone(&self.inner);
114        run_blocking(move || db.commit()).await
115    }
116
117    /// Rolls back the current explicit transaction.
118    pub async fn rollback(&self) -> Result<()> {
119        let db = Arc::clone(&self.inner);
120        run_blocking(move || db.rollback()).await
121    }
122
123    /// Returns all CDC changes with sequence numbers greater than `since`.
124    pub async fn changes_since(&self, since: u64) -> Vec<crate::cdc::ChangeEvent> {
125        let db = Arc::clone(&self.inner);
126        tokio::task::spawn_blocking(move || db.changes_since(since))
127            .await
128            .unwrap_or_default()
129    }
130
131    /// Synchronous escape hatch to the underlying [`Database`].
132    pub fn blocking(&self) -> &Database {
133        &self.inner
134    }
135}
136
137/// Runs `f` on the tokio blocking pool and flattens the join error.
138async fn run_blocking<T, F>(f: F) -> Result<T>
139where
140    T: Send + 'static,
141    F: FnOnce() -> Result<T> + Send + 'static,
142{
143    tokio::task::spawn_blocking(f)
144        .await
145        .map_err(|e| VelociError::IoError(format!("blocking task failed: {}", e)))?
146}

148#[cfg(test)]
149mod tests {
150    use super::*;
151    use tempfile::NamedTempFile;
152
153    #[tokio::test]
154    async fn test_async_roundtrip() {
155        let tmp = NamedTempFile::new().unwrap();
156        let db = Builder::new_local(tmp.path()).build().await.unwrap();
157        let conn = db.connect().unwrap();
158
159        conn.execute("CREATE TABLE t (id INTEGER PRIMARY KEY, name TEXT)")
160            .await
161            .unwrap();
162        conn.execute("INSERT INTO t VALUES (1, 'Alice')").await.unwrap();
163
164        let result = conn.query("SELECT * FROM t").await.unwrap();
165        assert_eq!(result.rows.len(), 1);
166    }
167
168    #[tokio::test]
169    async fn test_concurrent_async_readers() {
170        let tmp = NamedTempFile::new().unwrap();
171        let db = Builder::new_local(tmp.path()).build().await.unwrap();
172        let conn = db.connect().unwrap();
173
174        conn.execute("CREATE TABLE t (id INTEGER PRIMARY KEY, v INTEGER)")
175            .await
176            .unwrap();
177        for i in 0..50 {
178            conn.execute(&format!("INSERT INTO t VALUES ({}, {})", i, i * 2))
179                .await
180                .unwrap();
181        }
182
183        // Many concurrent read futures.
184        let mut handles = Vec::new();
185        for _ in 0..8 {
186            let c = db.connect().unwrap();
187            handles.push(tokio::spawn(async move {
188                c.query("SELECT * FROM t").await.unwrap().rows.len()
189            }));
190        }
191        for h in handles {
192            assert_eq!(h.await.unwrap(), 50);
193        }
194    }
