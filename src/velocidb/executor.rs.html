1//! Query executor that translates AST statements into storage operations.
2//!
3//! Coordinates B-Tree, MVCC, lock manager, and schema to execute SQL
4//! statements with ACID guarantees. WHERE filtering, ORDER BY sorting, and
5//! vector distance computation are parallelized with rayon for larger row
6//! sets.
7
8use crate::btree::BTree;
9use crate::cdc::{CdcManager, ChangeOp};
10use crate::parser::{AlterAction, Operator, OrderBy, Statement, WhereClause};
11use crate::storage::{Pager, Schema, TableSchema};
12use crate::transaction::{LockManager, LockType, Transaction, TransactionManager};

20/// Row-count threshold above which filtering/sorting switches to rayon.
21const PARALLEL_THRESHOLD: usize = 1024;

52    pub fn execute_statement(&self, statement: Statement) -> Result<()> {
53        match statement {
54            Statement::CreateTable { name, columns } => self.execute_create_table(&name, columns),
55            Statement::DropTable { name } => self.execute_drop_table(&name),
56            Statement::AlterTable { table, action } => self.execute_alter_table(&table, action),
57            Statement::Insert {
58                table,
59                columns,
60                values,
61            } => self.with_table_lock(&table, LockType::Exclusive, || {
62                self.execute_insert(&table, columns, values)
63            }),
64            Statement::Update {
65                table,
66                assignments,
67                where_clause,
68            } => self.with_table_lock(&table, LockType::Exclusive, || {
69                self.execute_update(&table, assignments, where_clause)
70            }),
71            Statement::Delete {
72                table,
73                where_clause,
74            } => self.with_table_lock(&table, LockType::Exclusive, || {
75                self.execute_delete(&table, where_clause)
76            }),
77            Statement::BeginTransaction => self.begin_transaction(),
78            Statement::CommitTransaction => self.commit_transaction(),
79            Statement::RollbackTransaction => self.rollback_transaction(),
80            _ => Err(VelociError::ParseError(
81                "Statement should be executed with query()".to_string(),
82            )),
83        }
84    }

86    pub fn query_statement(&self, statement: Statement) -> Result<QueryResult> {
87        match statement {
88            Statement::Select {
89                table,
90                columns,
91                where_clause,
92                order_by,
93                limit,
94            } => self.with_table_lock(&table, LockType::Shared, || {
95                self.execute_select(&table, columns, where_clause, order_by, limit)
96            }),
97            _ => Err(VelociError::ParseError(
98                "Statement is not a query".to_string(),
99            )),
100        }
101    }

103    /// Runs `f` holding a `lock` on `table`.
104    ///
105    /// Inside an explicit transaction the lock is taken on behalf of that
106    /// transaction and held until COMMIT / ROLLBACK, even if `f` fails (the
107    /// caller undoes the failed statement's storage writes). Otherwise a
108    /// single-statement transaction is started and always ended — committed
109    /// on success, aborted on error — with its lock released on every path.
110    fn with_table_lock<T>(
111        &self,
112        table: &str,
113        lock: LockType,
114        f: impl FnOnce() -> Result<T>,
115    ) -> Result<T> {
116        if let Some(txn) = self.active_transaction.read().clone() {
117            self.lock_manager.acquire_lock(table, txn.id(), lock)?;
118            return f();
119        }
120
121        let txn = self.transaction_manager.begin();
122        if let Err(e) = self.lock_manager.acquire_lock(table, txn.id(), lock) {
123            let _ = self.transaction_manager.abort(&txn);
124            return Err(e);
125        }
126        let result = f();
127        let ended = match result {
128            Ok(_) => self.transaction_manager.commit(&txn),
129            Err(_) => self.transaction_manager.abort(&txn),
130        };
131        self.lock_manager.release_all_locks(txn.id());
132        let value = result?;
133        ended?;
134        Ok(value)
135    }

137    /// Whether an explicit transaction is in progress.
138    pub fn in_transaction(&self) -> bool {
139        self.active_transaction.read().is_some()
140    }

142    pub fn begin_transaction(&self) -> Result<()> {
143        let mut active = self.active_transaction.write();
144        if active.is_some() {
145            return Err(VelociError::TransactionError(
146                "Transaction already in progress".to_string(),
147            ));
148        }
149        let txn = self.transaction_manager.begin();
150        *active = Some(txn);
151        Ok(())
152    }
153
154    pub fn commit_transaction(&self) -> Result<()> {
155        let mut active = self.active_transaction.write();
156        match active.take() {
157            Some(txn) => {
158                self.transaction_manager.commit(&txn)?;
159                self.lock_manager.release_all_locks(txn.id());
160                Ok(())
161            }
162            None => Err(VelociError::TransactionError(
163                "No active transaction to commit".to_string(),
164            )),
165        }
166    }
167
168    pub fn rollback_transaction(&self) -> Result<()> {
169        // Storage is undone by the caller (`Database::rollback` aborts the
170        // transaction's WAL group); this only ends the transaction and
171        // releases its locks.
172        let mut active = self.active_transaction.write();
173        match active.take() {
174            Some(txn) => {
175                self.transaction_manager.abort(&txn)?;
176                self.lock_manager.release_all_locks(txn.id());
177                Ok(())
178            }
179            None => Err(VelociError::TransactionError(
180                "No active transaction to rollback".to_string(),
181            )),
182        }
183    }

185    fn execute_create_table(&self, name: &str, columns: Vec<Column>) -> Result<()> {
186        // Allocate and initialize a root page for the table
187        let root_page = {
188            let mut pager = self.pager.write();
189            let root_page = pager.allocate_page()?;
190
191            // Initialize as B-Tree leaf node
192            let mut page = crate::storage::Page::new();
193            let header = crate::btree::NodeHeader::new_leaf();
194            header.serialize(page.data_mut());
195            pager.write_page(root_page, &page)?;
196
197            root_page
198        };
199
200        // Create the table schema
201        let table_schema = TableSchema {
202            name: name.to_string(),
203            columns,
204            root_page,
205        };
206
207        // Add to schema
208        self.schema.write().create_table(table_schema)?;
209
210        // Create a new B-Tree for the table
211        let btree = BTree::from_root(root_page, Arc::clone(&self.pager));
212        self.btrees
213            .write()
214            .insert(name.to_string(), Arc::new(RwLock::new(btree)));
215
216        Ok(())
217    }

219    fn execute_drop_table(&self, name: &str) -> Result<()> {
220        self.schema.write().drop_table(name)?;
221        self.btrees.write().remove(name);
222        Ok(())
223    }

225    fn execute_alter_table(&self, table: &str, action: AlterAction) -> Result<()> {
226        match action {
227            AlterAction::RenameTable { new_name } => {
228                {
229                    let schema = self.schema.read();
230                    if schema.get_table(&new_name).is_ok() {
231                        return Err(VelociError::ConstraintViolation(format!(
232                            "Table '{}' already exists",
233                            new_name
234                        )));
235                    }
236                }
237                let mut schema = self.schema.write();
238                let mut table_schema = schema.get_table(table)?.clone();
239                schema.drop_table(table)?;
240                table_schema.name = new_name.clone();
241                schema.create_table(table_schema)?;
242                drop(schema);
243
244                let mut btrees = self.btrees.write();
245                if let Some(bt) = btrees.remove(table) {
246                    btrees.insert(new_name, bt);
247                }
248                Ok(())
249            }
250            AlterAction::RenameColumn { old_name, new_name } => {
251                let mut schema = self.schema.write();
252                let table_schema = schema.get_table_mut(table)?;
253                if table_schema.columns.iter().any(|c| c.name == new_name) {
254                    return Err(VelociError::ConstraintViolation(format!(
255                        "Column '{}' already exists in table '{}'",
256                        new_name, table
257                    )));
258                }
259                let col = table_schema
260                    .columns
261                    .iter_mut()
262                    .find(|c| c.name == old_name)
263                    .ok_or_else(|| {
264                        VelociError::NotFound(format!(
265                            "Column '{}' not found in table '{}'",
266                            old_name, table
267                        ))
268                    })?;
269                col.name = new_name;
270                Ok(())
271            }
272            AlterAction::AddColumn { column } => {
273                {
274                    let schema = self.schema.read();
275                    let table_schema = schema.get_table(table)?;
276                    if table_schema.columns.iter().any(|c| c.name == column.name) {
277                        return Err(VelociError::ConstraintViolation(format!(
278                            "Column '{}' already exists in table '{}'",
279                            column.name, table
280                        )));
281                    }
282                }
283
284                // Rewrite existing rows with a trailing NULL so row width
285                // matches the new schema.
286                {
287                    let btrees = self.btrees.read();
288                    let btree_arc = btrees.get(table).ok_or_else(|| {
289                        VelociError::NotFound(format!("Table '{}' not initialized", table))
290                    })?;
291                    let mut btree = btree_arc.write();
292                    let all_rows = btree.scan()?;
293                    for (key, mut row) in all_rows {
294                        row.values.push(Value::Null);
295                        btree.delete(key)?;
296                        btree.insert(key, &row)?;
297                    }
298                }
299
300                self.schema
301                    .write()
302                    .get_table_mut(table)?
303                    .columns
304                    .push(column);
305                Ok(())
306            }
307            AlterAction::DropColumn { name } => {
308                let col_index = {
309                    let schema = self.schema.read();
310                    let table_schema = schema.get_table(table)?;
311                    let idx = table_schema
312                        .columns
313                        .iter()
314                        .position(|c| c.name == name)
315                        .ok_or_else(|| {
316                            VelociError::NotFound(format!(
317                                "Column '{}' not found in table '{}'",
318                                name, table
319                            ))
320                        })?;
321                    if table_schema.columns[idx].primary_key {
322                        return Err(VelociError::ConstraintViolation(
323                            "Cannot drop the primary key column".to_string(),
324                        ));
325                    }
326                    idx
327                };
328
329                {
330                    let btrees = self.btrees.read();
331                    let btree_arc = btrees.get(table).ok_or_else(|| {
332                        VelociError::NotFound(format!("Table '{}' not initialized", table))
333                    })?;
334                    let mut btree = btree_arc.write();
335                    let all_rows = btree.scan()?;
336                    for (key, mut row) in all_rows {
337                        if col_index < row.values.len() {
338                            row.values.remove(col_index);
339                        }
340                        btree.delete(key)?;
341                        btree.insert(key, &row)?;
342                    }
343                }
344
345                self.schema
346                    .write()
347                    .get_table_mut(table)?
348                    .columns
349                    .remove(col_index);
350                Ok(())
351            }
352        }
353    }

355    fn execute_insert(
356        &self,
357        table: &str,
358        columns: Option<Vec<String>>,
359        values: Vec<Value>,
360    ) -> Result<()> {
361        // Get table schema and immediately clone to release lock
362        let table_schema = {
363            let schema = self.schema.read();
364            schema.get_table(table)?.clone()
365        }; // schema lock released here
366
367        // Validate columns and values
368        let column_names = if let Some(ref cols) = columns {
369            cols.clone()
370        } else {
371            table_schema
372                .columns
373                .iter()
374                .map(|c| c.name.clone())
375                .collect()
376        };
377
378        if column_names.len() != values.len() {
379            return Err(VelociError::ConstraintViolation(
380                "Column count doesn't match value count".to_string(),
381            ));
382        }
383
384        // Find primary key
385        let pk_index = table_schema
386            .columns
387            .iter()
388            .position(|c| c.primary_key)
389            .ok_or_else(|| {
390                VelociError::ConstraintViolation("No primary key defined".to_string())
391            })?;
392
393        let pk_col_name = &table_schema.columns[pk_index].name;
394        let pk_value_index = column_names
395            .iter()
396            .position(|c| c == pk_col_name)
397            .ok_or_else(|| {
398                VelociError::ConstraintViolation("Primary key value not provided".to_string())
399            })?;
400
401        let pk_value = values[pk_value_index].as_integer()?;
402
403        // Validate NOT NULL and vector-dimension constraints before acquiring
404        // the BTree lock.
405        for (i, value) in values.iter().enumerate() {
406            let col_name = &column_names[i];
407            if let Some(col) = table_schema.columns.iter().find(|c| &c.name == col_name) {
408                if col.not_null && matches!(value, Value::Null) {
409                    return Err(VelociError::ConstraintViolation(format!(
410                        "Column '{}' cannot be NULL",
411                        col_name
412                    )));
413                }
414                if let DataType::Vector(dim) = col.data_type {
415                    match value {
416                        Value::Null => {}
417                        Value::Vector(v) if v.len() == dim as usize => {}
418                        Value::Vector(v) => {
419                            return Err(VelociError::ConstraintViolation(format!(
420                                "Column '{}' expects a vector of dimension {}, got {}",
421                                col_name,
422                                dim,
423                                v.len()
424                            )));
425                        }
426                        other => {
427                            return Err(VelociError::TypeMismatch {
428                                expected: format!("Vector({})", dim),
429                                actual: format!("{:?}", other),
430                            });
431                        }
432                    }
433                }
434            }
435        }
436
437        // Create row data
438        let mut row_values = vec![Value::Null; table_schema.columns.len()];
439        for (i, col_name) in column_names.iter().enumerate() {
440            if let Some(col_index) = table_schema
441                .columns
442                .iter()
443                .position(|c| &c.name == col_name)
444            {
445                row_values[col_index] = values[i].clone();
446            }
447        }
448        let row = Row::new(row_values);
449
450        // Single acquisition of btrees collection and btree with write intent
451        // No lock upgrade, no multiple acquisitions
452        let result = {
453            let btrees = self.btrees.read();
454            let btree_arc = btrees.get(table).ok_or_else(|| {
455                VelociError::NotFound(format!("Table '{}' not initialized", table))
456            })?;
457
458            // Acquire write lock directly (no upgrade from read)
459            let mut btree = btree_arc.write();
460
461            // Check for primary key uniqueness
462            if btree.search(pk_value)?.is_some() {
463                Err(VelociError::ConstraintViolation(format!(
464                    "Primary key {} already exists in table '{}'",
465                    pk_value, table
466                )))
467            } else if let Err(e) = check_insert_unique(&table_schema, &btree, &row) {
468                Err(e)
469            } else {
470                // Insert into B-Tree for persistence
471                btree.insert(pk_value, &row)?;
472                Ok(())
473            }
474        }; // btrees and btree locks released here
475
476        // Handle result
477        result?;
478
479        self.cdc
480            .stage(table, ChangeOp::Insert, pk_value, None, Some(row));
481
482        // Only commit/release lock for auto-commit mode
483
484        Ok(())
485    }

487    fn execute_select(
488        &self,
489        table: &str,
490        columns: Vec<String>,
491        where_clause: Option<WhereClause>,
492        order_by: Option<OrderBy>,
493        limit: Option<u64>,
494    ) -> Result<QueryResult> {
495        // Get table schema and release lock immediately
496        let table_schema = {
497            let schema = self.schema.read();
498            schema.get_table(table)?.clone()
499        }; // schema lock released
500
501        // The B-tree is authoritative. MVCC overlay is intentionally not used
502        // until WAL-backed snapshot isolation lands.
503        let all_rows: Vec<(i64, Row)> = {
504            let btrees = self.btrees.read();
505            let btree_arc = btrees.get(table).ok_or_else(|| {
506                VelociError::NotFound(format!("Table '{}' not initialized", table))
507            })?;
508            let btree = btree_arc.read();
509            candidate_rows(&btree, where_clause.as_ref(), &table_schema)?
510        };
511
512        // Process data without holding any locks. Filtering runs in parallel
513        // (rayon) once the row count crosses PARALLEL_THRESHOLD.
514        let mut filtered_rows: Vec<(i64, Row)> = if let Some(ref where_clause) = where_clause {
515            if all_rows.len() >= PARALLEL_THRESHOLD {
516                all_rows
517                    .into_par_iter()
518                    .filter(|(_, row)| {
519                        self.evaluate_where_clause(row, where_clause, &table_schema)
520                            .unwrap_or(false)
521                    })
522                    .collect()
523            } else {
524                all_rows
525                    .into_iter()
526                    .filter(|(_, row)| {
527                        self.evaluate_where_clause(row, where_clause, &table_schema)
528                            .unwrap_or(false)
529                    })
530                    .collect()
531            }
532        } else {
533            all_rows
534        };
535
536        // Apply ORDER BY. Two forms are supported:
537        // 1. A vector distance expression -> exact KNN (parallel distances).
538        // 2. A plain column -> comparison sort (parallel for large sets).
539        if let Some(ref order) = order_by {
540            if let Some(parsed) = vector::parse_distance_expr(&order.column) {
541                let expr = parsed?;
542                let col_index = table_schema
543                    .columns
544                    .iter()
545                    .position(|c| c.name == expr.column)
546                    .ok_or_else(|| {
547                        VelociError::NotFound(format!(
548                            "Vector column '{}' not found in table '{}'",
549                            expr.column, table
550                        ))
551                    })?;
552
553                if order.ascending {
554                    // Nearest-first with a LIMIT is the classic KNN shape:
555                    // use top-k selection instead of a full sort.
556                    let k = limit
557                        .map(|n| n as usize)
558                        .unwrap_or(usize::MAX)
559                        .min(filtered_rows.len());
560                    filtered_rows =
561                        vector::knn(filtered_rows, col_index, &expr.query, expr.metric, k)
562                            .into_iter()
563                            .map(|(_, key, row)| (key, row))
564                            .collect();
565                } else {
566                    let distances = vector::compute_distances(
567                        &filtered_rows,
568                        col_index,
569                        &expr.query,
570                        expr.metric,
571                    );
572                    let mut scored: Vec<(f64, i64, Row)> = filtered_rows
573                        .into_iter()
574                        .zip(distances)
575                        .map(|((key, row), d)| (d, key, row))
576                        .collect();
577                    scored.sort_by(|a, b| {
578                        b.0.partial_cmp(&a.0)
579                            .unwrap_or(std::cmp::Ordering::Equal)
580                            .then(a.1.cmp(&b.1))
581                    });
582                    filtered_rows = scored.into_iter().map(|(_, key, row)| (key, row)).collect();
583                }
584            } else {
585                let col_index = table_schema
586                    .columns
587                    .iter()
588                    .position(|c| c.name == order.column)
589                    .ok_or_else(|| {
590                        VelociError::NotFound(format!(
591                            "ORDER BY column '{}' not found in table '{}'",
592                            order.column, table
593                        ))
594                    })?;
595
596                let cmp = |(ak, a): &(i64, Row), (bk, b): &(i64, Row)| {
597                    let av = a.values.get(col_index).unwrap_or(&Value::Null);
598                    let bv = b.values.get(col_index).unwrap_or(&Value::Null);
599                    let primary = compare_values(av, bv);
600                    let secondary = ak.cmp(bk);
601                    let combined = primary.then(secondary);
602                    if order.ascending {
603                        combined
604                    } else {
605                        combined.reverse()
606                    }
607                };
608
609                if filtered_rows.len() >= PARALLEL_THRESHOLD {
610                    filtered_rows.par_sort_by(cmp);
611                } else {
612                    filtered_rows.sort_by(cmp);
613                }
614            }
615        }
616
617        // Apply LIMIT n (does not affect COUNT(*), which counts rows after WHERE).
618        if let Some(n) = limit {
619            filtered_rows.truncate(n as usize);
620        }
621
622        // Check for aggregate functions (COUNT)
623        let is_count = columns.len() == 1 && columns[0].to_uppercase().starts_with("COUNT(");
624
625        if is_count {
626            let count = filtered_rows.len() as i64;
627            let result_columns = vec![Column {
628                name: columns[0].clone(),
629                data_type: crate::types::DataType::Integer,
630                primary_key: false,
631                not_null: true,
632                unique: false,
633            }];
634            let result_rows = vec![Row::new(vec![Value::Integer(count)])];
635
636            return Ok(QueryResult::new(result_columns, result_rows));
637        }
638
639        // Project columns. Each projection item is either '*', a table
640        // column, or a vector distance expression computed per row.
641        enum ProjItem {
642            Star,
643            Col(usize),
644            Distance {
645                col_index: usize,
646                metric: DistanceMetric,
647                query: Vec<f32>,
648            },
649        }
650
651        let mut proj_items: Vec<(String, ProjItem)> = Vec::with_capacity(columns.len());
652        for col_name in &columns {
653            if col_name == "*" {
654                proj_items.push(("*".to_string(), ProjItem::Star));
655            } else if let Some(idx) = table_schema
656                .columns
657                .iter()
658                .position(|c| &c.name == col_name)
659            {
660                proj_items.push((col_name.clone(), ProjItem::Col(idx)));
661            } else if let Some(parsed) = vector::parse_distance_expr(col_name) {
662                let expr = parsed?;
663                let col_index = table_schema
664                    .columns
665                    .iter()
666                    .position(|c| c.name == expr.column)
667                    .ok_or_else(|| {
668                        VelociError::NotFound(format!(
669                            "Vector column '{}' not found in table '{}'",
670                            expr.column, table
671                        ))
672                    })?;
673                proj_items.push((
674                    col_name.clone(),
675                    ProjItem::Distance {
676                        col_index,
677                        metric: expr.metric,
678                        query: expr.query,
679                    },
680                ));
681            } else {
682                return Err(VelociError::NotFound(format!(
683                    "Column '{}' not found in table '{}'",
684                    col_name, table
685                )));
686            }
687        }
688
689        let mut result_columns: Vec<Column> = Vec::new();
690        for (name, item) in &proj_items {
691            match item {
692                ProjItem::Star => result_columns.extend(table_schema.columns.iter().cloned()),
693                ProjItem::Col(idx) => result_columns.push(table_schema.columns[*idx].clone()),
694                ProjItem::Distance { .. } => result_columns.push(Column {
695                    name: name.clone(),
696                    data_type: DataType::Real,
697                    primary_key: false,
698                    not_null: false,
699                    unique: false,
700                }),
701            }
702        }
703
704        let result_rows: Vec<Row> = filtered_rows
705            .into_iter()
706            .map(|(_, row)| {
707                let mut projected: Vec<Value> = Vec::with_capacity(result_columns.len());
708                for (_, item) in &proj_items {
709                    match item {
710                        ProjItem::Star => projected.extend(row.values.iter().cloned()),
711                        ProjItem::Col(idx) => {
712                            projected.push(row.values.get(*idx).cloned().unwrap_or(Value::Null))
713                        }
714                        ProjItem::Distance {
715                            col_index,
716                            metric,
717                            query,
718                        } => {
719                            let d = row
720                                .values
721                                .get(*col_index)
722                                .and_then(|v| vector::value_as_vector(v).ok())
723                                .and_then(|v| metric.distance(&v, query).ok());
724                            projected.push(match d {
725                                Some(d) => Value::Float(d),
726                                None => Value::Null,
727                            });
728                        }
729                    }
730                }
731                Row::new(projected)
732            })
733            .collect();
734
735        Ok(QueryResult::new(result_columns, result_rows))
736    }

738    fn execute_update(
739        &self,
740        table: &str,
741        assignments: HashMap<String, Value>,
742        where_clause: Option<WhereClause>,
743    ) -> Result<()> {
744        // Get table schema and release lock
745        let table_schema = {
746            let schema = self.schema.read();
747            schema.get_table(table)?.clone()
748        }; // schema lock released
749
750        // Change events collected during the update, recorded to CDC only
751        // after the whole statement succeeds.
752        let mut cdc_events: Vec<(i64, Row, Row)> = Vec::new();
753
754        // Perform update with single btree lock acquisition
755        let result = {
756            let btrees = self.btrees.read();
757            let btree_arc = btrees.get(table).ok_or_else(|| {
758                VelociError::NotFound(format!("Table '{}' not initialized", table))
759            })?;
760            let mut btree = btree_arc.write();
761
762            let all_rows = candidate_rows(&btree, where_clause.as_ref(), &table_schema)?;
763
764            // Find rows to update
765            let rows_to_update: Vec<(i64, Row)> = if let Some(ref where_clause) = where_clause {
766                all_rows
767                    .into_iter()
768                    .filter(|(_, row)| {
769                        self.evaluate_where_clause(row, where_clause, &table_schema)
770                            .unwrap_or(false)
771                    })
772                    .collect()
773            } else {
774                all_rows
775            };
776
777            // Find primary key column
778            let pk_index = table_schema
779                .columns
780                .iter()
781                .position(|c| c.primary_key)
782                .ok_or_else(|| {
783                    VelociError::ConstraintViolation("No primary key defined".to_string())
784                })?;
785
786            // Compute every updated row first so UNIQUE can be checked
787            // against the table's final state before anything is mutated.
788            let mut updates: Vec<(i64, i64, Row, Row)> = Vec::with_capacity(rows_to_update.len());
789            for (key, row) in &rows_to_update {
790                let mut updated_row = row.clone();
791                let mut new_pk_value = *key; // Default to existing key
792                let mut pk_being_updated = false;
793
794                // Apply updates to the row
795                for (col_name, new_value) in &assignments {
796                    if let Some(col_index) = table_schema
797                        .columns
798                        .iter()
799                        .position(|c| &c.name == col_name)
800                    {
801                        // Check NOT NULL constraint
802                        let col = &table_schema.columns[col_index];
803                        if col.not_null && matches!(new_value, Value::Null) {
804                            return Err(VelociError::ConstraintViolation(format!(
805                                "Column '{}' cannot be NULL",
806                                col_name
807                            )));
808                        }
809
810                        updated_row.values[col_index] = new_value.clone();
811
812                        // Check if primary key is being updated
813                        if col_index == pk_index {
814                            new_pk_value = new_value.as_integer()?;
815                            pk_being_updated = true;
816                        }
817                    }
818                }
819
820                // If primary key is being updated, check for uniqueness
821                if pk_being_updated && new_pk_value != *key && btree.search(new_pk_value)?.is_some()
822                {
823                    return Err(VelociError::ConstraintViolation(format!(
824                        "Primary key {} already exists in table '{}'",
825                        new_pk_value, table
826                    )));
827                }
828
829                updates.push((*key, new_pk_value, row.clone(), updated_row));
830            }
831
832            let touched_unique: Vec<usize> = unique_columns(&table_schema)
833                .into_iter()
834                .filter(|&ci| assignments.contains_key(&table_schema.columns[ci].name))
835                .collect();
836            if !touched_unique.is_empty() && !updates.is_empty() {
837                let updated_keys: std::collections::HashSet<i64> =
838                    updates.iter().map(|(key, ..)| *key).collect();
839                let untouched = btree.scan()?;
840                let final_rows: Vec<&Row> = untouched
841                    .iter()
842                    .filter(|(key, _)| !updated_keys.contains(key))
843                    .map(|(_, row)| row)
844                    .chain(updates.iter().map(|(_, _, _, updated)| updated))
845                    .collect();
846                check_unique(&table_schema, &touched_unique, &final_rows)?;
847            }
848
849            // Delete old row and insert updated row
850            for (key, new_pk_value, row, updated_row) in updates {
851                btree.delete(key)?;
852                btree.insert(new_pk_value, &updated_row)?;
853
854                if self.cdc.is_enabled() {
855                    cdc_events.push((key, row, updated_row));
856                }
857            }
858
859            Ok::<(), VelociError>(())
860        }; // btrees and btree locks released
861
862        // Handle errors
863        result?;
864
865        for (key, before, after) in cdc_events {
866            self.cdc
867                .stage(table, ChangeOp::Update, key, Some(before), Some(after));
868        }
869
870        Ok(())
871    }

873    #[allow(clippy::let_and_return)]
874    fn execute_delete(&self, table: &str, where_clause: Option<WhereClause>) -> Result<()> {
875        // Get table schema and release lock
876        let table_schema = {
877            let schema = self.schema.read();
878            schema.get_table(table)?.clone()
879        }; // schema lock released
880
881        // Perform delete with single btree lock acquisition
882        let mut cdc_events: Vec<(i64, Row)> = Vec::new();
883        let result: Result<()> = {
884            let btrees = self.btrees.read();
885            let btree_arc = btrees.get(table).ok_or_else(|| {
886                VelociError::NotFound(format!("Table '{}' not initialized", table))
887            })?;
888            let mut btree = btree_arc.write();
889
890            let all_rows = candidate_rows(&btree, where_clause.as_ref(), &table_schema)?;
891            let rows_to_delete: Vec<(i64, Row)> = if let Some(ref where_clause) = where_clause {
892                all_rows
893                    .into_iter()
894                    .filter(|(_, row)| {
895                        self.evaluate_where_clause(row, where_clause, &table_schema)
896                            .unwrap_or(false)
897                    })
898                    .collect()
899            } else {
900                all_rows
901            };
902
903            for (key, row) in rows_to_delete {
904                btree.delete(key)?;
905                if self.cdc.is_enabled() {
906                    cdc_events.push((key, row));
907                }
908            }
909            Ok(())
910        };
911
912        result?;
913
914        for (key, before) in cdc_events {
915            self.cdc
916                .stage(table, ChangeOp::Delete, key, Some(before), None);
917        }
918
919        Ok(())
920    }

922    fn evaluate_where_clause(
923        &self,
924        row: &Row,
925        where_clause: &WhereClause,
926        table_schema: &TableSchema,
927    ) -> Result<bool> {
928        // (helper) See `compare_values` below for sort ordering semantics.
929        for condition in &where_clause.conditions {
930            let col_index = table_schema
931                .columns
932                .iter()
933                .position(|c| c.name == condition.column)
934                .ok_or_else(|| {
935                    VelociError::NotFound(format!("Column '{}' not found", condition.column))
936                })?;
937
938            let row_value = &row.values[col_index];
939            let condition_value = &condition.value;
940
941            if !condition.operator.evaluate(row_value, condition_value)? {
942                return Ok(false);
943            }
944        }
945
946        Ok(true)
947    }
948}

950/// Order two `Value`s for ORDER BY purposes.
951///
952/// SQL NULL semantics here: NULL sorts after all non-NULL values (the SQLite
953/// default for ASC and the conventional "NULLS LAST" behaviour).
954fn compare_values(a: &Value, b: &Value) -> std::cmp::Ordering {
955    use std::cmp::Ordering;
956    match (a, b) {
957        (Value::Null, Value::Null) => Ordering::Equal,
958        (Value::Null, _) => Ordering::Greater,
959        (_, Value::Null) => Ordering::Less,
960        (Value::Integer(x), Value::Integer(y)) => x.cmp(y),
961        (Value::Text(x), Value::Text(y)) => x.cmp(y),
962        (Value::Blob(x), Value::Blob(y)) => x.cmp(y),
963        // Mixed numeric: promote to f64.
964        (lhs, rhs)
965            if matches!(lhs, Value::Float(_) | Value::Real(_) | Value::Integer(_))
966                && matches!(rhs, Value::Float(_) | Value::Real(_) | Value::Integer(_)) =>
967        {
968            let lf = lhs.as_float().unwrap_or(0.0);
969            let rf = rhs.as_float().unwrap_or(0.0);
970            lf.partial_cmp(&rf).unwrap_or(Ordering::Equal)
971        }
972        // Different incomparable types: fall back to discriminant order so that
973        // the sort is at least total and deterministic.
974        _ => format!("{:?}", a).cmp(&format!("{:?}", b)),
975    }
976}

978/// Rows that can possibly match `where_clause`: the single row found by a
979/// B-tree lookup when the clause contains `<pk> = <integer>`, otherwise every
980/// row. Callers still evaluate the full clause on the result.
981fn candidate_rows(
982    btree: &BTree,
983    where_clause: Option<&WhereClause>,
984    table: &TableSchema,
985) -> Result<Vec<(i64, Row)>> {
986    let pk_name = table
987        .columns
988        .iter()
989        .find(|c| c.primary_key)
990        .map(|c| c.name.as_str());
991    let pk_key = where_clause.zip(pk_name).and_then(|(wc, pk)| {
992        wc.conditions
993            .iter()
994            .find_map(|c| match (&c.operator, &c.value) {
995                (Operator::Equal, Value::Integer(key)) if c.column == pk => Some(*key),
996                _ => None,
997            })
998    });
999    match pk_key {
1000        Some(key) => Ok(btree
1001            .search(key)?
1002            .map(|row| (key, row))
1003            .into_iter()
1004            .collect()),
1005        None => btree.scan(),
1006    }
1007}

1009/// Indices of `UNIQUE` columns other than the primary key (the B-tree
1010/// enforces primary-key uniqueness itself).
1011fn unique_columns(table: &TableSchema) -> Vec<usize> {
1012    table
1013        .columns
1014        .iter()
1015        .enumerate()
1016        .filter(|(_, c)| c.unique && !c.primary_key)
1017        .map(|(i, _)| i)
1018        .collect()
1019}

1021fn unique_violation(table: &TableSchema, col_index: usize, value: &Value) -> VelociError {
1022    VelociError::ConstraintViolation(format!(
1023        "UNIQUE constraint failed: {}.{} (duplicate value {})",
1024        table.name, table.columns[col_index].name, value
1025    ))
1026}

1028/// Rejects `row` if a UNIQUE column's non-NULL value already exists in the
1029/// table. Without secondary indexes this is a full scan, done only when the
1030/// table has UNIQUE columns.
1031fn check_insert_unique(table: &TableSchema, btree: &BTree, row: &Row) -> Result<()> {
1032    let cols: Vec<usize> = unique_columns(table)
1033        .into_iter()
1034        .filter(|&ci| !matches!(row.values.get(ci), None | Some(Value::Null)))
1035        .collect();
1036    if cols.is_empty() {
1037        return Ok(());
1038    }
1039    for (_, existing) in btree.scan()? {
1040        for &ci in &cols {
1041            if let Some(v) = existing.values.get(ci) {
1042                if !matches!(v, Value::Null)
1043                    && compare_values(v, &row.values[ci]) == std::cmp::Ordering::Equal
1044                {
1045                    return Err(unique_violation(table, ci, v));
1046                }
1047            }
1048        }
1049    }
1050    Ok(())
1051}
1052
1053/// Rejects `rows` (a table's complete final contents) if any column in
1054/// `cols` holds the same non-NULL value twice. NULLs never conflict.
1055fn check_unique(table: &TableSchema, cols: &[usize], rows: &[&Row]) -> Result<()> {
1056    for &ci in cols {
1057        let mut values: Vec<&Value> = rows
1058            .iter()
1059            .filter_map(|r| r.values.get(ci))
1060            .filter(|v| !matches!(v, Value::Null))
1061            .collect();
1062        values.sort_by(|a, b| compare_values(a, b));
1063        if let Some(pair) = values
1064            .windows(2)
1065            .find(|w| compare_values(w[0], w[1]) == std::cmp::Ordering::Equal)
1066        {
1067            return Err(unique_violation(table, ci, pair[0]));
1068        }
1069    }
1070    Ok(())
1071}
