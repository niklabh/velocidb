1//! Query executor that translates AST statements into storage operations.
2//!
3//! Coordinates B-Tree, MVCC, lock manager, and schema to execute SQL
4//! statements with ACID guarantees. WHERE filtering, ORDER BY sorting, and
5//! vector distance computation are parallelized with rayon for larger row
6//! sets.
7
8use crate::btree::BTree;
9use crate::cdc::{CdcManager, ChangeOp};
10use crate::parser::{AlterAction, OrderBy, Statement, WhereClause};
11use crate::storage::{Pager, Schema, TableSchema};
12use crate::transaction::{LockManager, LockType, TransactionManager, Transaction};
13use crate::types::{Column, DataType, QueryResult, Result, Row, Value, VelociError};
14use crate::vector::{self, DistanceMetric};
15use parking_lot::RwLock;
16use rayon::prelude::*;
17use std::collections::HashMap;
18use std::sync::Arc;
19
20/// Row-count threshold above which filtering/sorting switches to rayon.
21const PARALLEL_THRESHOLD: usize = 1024;
22
23pub struct Executor {
24    pager: Arc<RwLock<Pager>>,
25    btrees: Arc<RwLock<HashMap<String, Arc<RwLock<BTree>>>>>,
26    schema: Arc<RwLock<Schema>>,
27    transaction_manager: Arc<TransactionManager>,
28    lock_manager: Arc<LockManager>,
29    active_transaction: RwLock<Option<Arc<Transaction>>>,
30    cdc: Arc<CdcManager>,
31}

33impl Executor {
34    pub fn new(
35        pager: Arc<RwLock<Pager>>,
36        btrees: Arc<RwLock<HashMap<String, Arc<RwLock<BTree>>>>>,
37        schema: Arc<RwLock<Schema>>,
38        transaction_manager: Arc<TransactionManager>,
39        cdc: Arc<CdcManager>,
40    ) -> Self {
41        Self {
42            pager,
43            btrees,
44            schema,
45            transaction_manager,
46            lock_manager: Arc::new(LockManager::new()),
47            active_transaction: RwLock::new(None),
48            cdc,
49        }

52    pub fn execute_statement(&self, statement: Statement) -> Result<()> {
53        match statement {
54            Statement::CreateTable { name, columns } => self.execute_create_table(&name, columns),
55            Statement::DropTable { name } => self.execute_drop_table(&name),
56            Statement::AlterTable { table, action } => self.execute_alter_table(&table, action),
57            Statement::Insert { table, columns, values } => self.execute_insert(&table, columns, values),
58            Statement::Update { table, assignments, where_clause } => self.execute_update(&table, assignments, where_clause),
59            Statement::Delete { table, where_clause } => self.execute_delete(&table, where_clause),
60            Statement::BeginTransaction => self.begin_transaction(),
61            Statement::CommitTransaction => self.commit_transaction(),
62            Statement::RollbackTransaction => self.rollback_transaction(),
63            _ => Err(VelociError::ParseError("Statement should be executed with query()".to_string())),
64        }
65    }
66
67    pub fn query_statement(&self, statement: Statement) -> Result<QueryResult> {
68        match statement {
69            Statement::Select { table, columns, where_clause, order_by, limit } => {
70                self.execute_select(&table, columns, where_clause, order_by, limit)
71            }
72            _ => Err(VelociError::ParseError("Statement is not a query".to_string())),
73        }
74    }

76    pub fn begin_transaction(&self) -> Result<()> {
77        let mut active = self.active_transaction.write();
78        if active.is_some() {
79            return Err(VelociError::TransactionError(
80                "Transaction already in progress".to_string(),
81            ));
82        }
83        let txn = self.transaction_manager.begin();
84        *active = Some(txn);
85        Ok(())
86    }
87
88    pub fn commit_transaction(&self) -> Result<()> {
89        let mut active = self.active_transaction.write();
90        match active.take() {
91            Some(txn) => {
92                self.transaction_manager.commit(&txn)?;
93                self.lock_manager.release_all_locks(txn.id());
94                Ok(())
95            }
96            None => Err(VelociError::TransactionError(
97                "No active transaction to commit".to_string(),
98            )),
99        }
100    }

102    pub fn rollback_transaction(&self) -> Result<()> {
103        // NOTE: Until WAL undo is wired in, rolling back an explicit
104        // transaction only releases locks. Any storage mutations the
105        // transaction performed remain on disk. The WAL milestone will close
106        // this gap.
107        let mut active = self.active_transaction.write();
108        match active.take() {
109            Some(txn) => {
110                self.transaction_manager.abort(&txn)?;
111                self.lock_manager.release_all_locks(txn.id());
112                Ok(())
113            }
114            None => Err(VelociError::TransactionError(
115                "No active transaction to rollback".to_string(),
116            )),
117        }
118    }

120    fn execute_create_table(&self, name: &str, columns: Vec<Column>) -> Result<()> {
121        // Allocate and initialize a root page for the table
122        let root_page = {
123            let mut pager = self.pager.write();
124            let root_page = pager.allocate_page()?;
125            
126            // Initialize as B-Tree leaf node
127            let mut page = crate::storage::Page::new();
128            let header = crate::btree::NodeHeader::new_leaf();
129            header.serialize(page.data_mut());
130            pager.write_page(root_page, &page)?;
131            
132            root_page
133        };
134
135        // Create the table schema
136        let table_schema = TableSchema {
137            name: name.to_string(),
138            columns,
139            root_page,
140        };
141
142        // Add to schema
143        self.schema.write().create_table(table_schema)?;
144
145        // Create a new B-Tree for the table
146        let btree = BTree::from_root(root_page, Arc::clone(&self.pager));
147        self.btrees
148            .write()
149            .insert(name.to_string(), Arc::new(RwLock::new(btree)));
150
151        Ok(())
152    }
153
154    fn execute_drop_table(&self, name: &str) -> Result<()> {
155        self.schema.write().drop_table(name)?;
156        self.btrees.write().remove(name);
157        Ok(())
158    }

160    fn execute_alter_table(&self, table: &str, action: AlterAction) -> Result<()> {
161        match action {
162            AlterAction::RenameTable { new_name } => {
163                {
164                    let schema = self.schema.read();
165                    if schema.get_table(&new_name).is_ok() {
166                        return Err(VelociError::ConstraintViolation(format!(
167                            "Table '{}' already exists",
168                            new_name
169                        )));
170                    }
171                }
172                let mut schema = self.schema.write();
173                let mut table_schema = schema.get_table(table)?.clone();
174                schema.drop_table(table)?;
175                table_schema.name = new_name.clone();
176                schema.create_table(table_schema)?;
177                drop(schema);
178
179                let mut btrees = self.btrees.write();
180                if let Some(bt) = btrees.remove(table) {
181                    btrees.insert(new_name, bt);
182                }
183                Ok(())
184            }
185            AlterAction::RenameColumn { old_name, new_name } => {
186                let mut schema = self.schema.write();
187                let table_schema = schema.get_table_mut(table)?;
188                if table_schema.columns.iter().any(|c| c.name == new_name) {
189                    return Err(VelociError::ConstraintViolation(format!(
190                        "Column '{}' already exists in table '{}'",
191                        new_name, table
192                    )));
193                }
194                let col = table_schema
195                    .columns
196                    .iter_mut()
197                    .find(|c| c.name == old_name)
198                    .ok_or_else(|| {
199                        VelociError::NotFound(format!(
200                            "Column '{}' not found in table '{}'",
201                            old_name, table
202                        ))
203                    })?;
204                col.name = new_name;
205                Ok(())
206            }
207            AlterAction::AddColumn { column } => {
208                {
209                    let schema = self.schema.read();
210                    let table_schema = schema.get_table(table)?;
211                    if table_schema.columns.iter().any(|c| c.name == column.name) {
212                        return Err(VelociError::ConstraintViolation(format!(
213                            "Column '{}' already exists in table '{}'",
214                            column.name, table
215                        )));
216                    }
217                }
218
219                // Rewrite existing rows with a trailing NULL so row width
220                // matches the new schema.
221                {
222                    let btrees = self.btrees.read();
223                    let btree_arc = btrees.get(table).ok_or_else(|| {
224                        VelociError::NotFound(format!("Table '{}' not initialized", table))
225                    })?;
226                    let mut btree = btree_arc.write();
227                    let all_rows = btree.scan()?;
228                    for (key, mut row) in all_rows {
229                        row.values.push(Value::Null);
230                        btree.delete(key)?;
231                        btree.insert(key, &row)?;
232                    }
233                }
234
235                self.schema
236                    .write()
237                    .get_table_mut(table)?
238                    .columns
239                    .push(column);
240                Ok(())
241            }
242            AlterAction::DropColumn { name } => {
243                let col_index = {
244                    let schema = self.schema.read();
245                    let table_schema = schema.get_table(table)?;
246                    let idx = table_schema
247                        .columns
248                        .iter()
249                        .position(|c| c.name == name)
250                        .ok_or_else(|| {
251                            VelociError::NotFound(format!(
252                                "Column '{}' not found in table '{}'",
253                                name, table
254                            ))
255                        })?;
256                    if table_schema.columns[idx].primary_key {
257                        return Err(VelociError::ConstraintViolation(
258                            "Cannot drop the primary key column".to_string(),
259                        ));
260                    }
261                    idx
262                };
263
264                {
265                    let btrees = self.btrees.read();
266                    let btree_arc = btrees.get(table).ok_or_else(|| {
267                        VelociError::NotFound(format!("Table '{}' not initialized", table))
268                    })?;
269                    let mut btree = btree_arc.write();
270                    let all_rows = btree.scan()?;
271                    for (key, mut row) in all_rows {
272                        if col_index < row.values.len() {
273                            row.values.remove(col_index);
274                        }
275                        btree.delete(key)?;
276                        btree.insert(key, &row)?;
277                    }
278                }
279
280                self.schema
281                    .write()
282                    .get_table_mut(table)?
283                    .columns
284                    .remove(col_index);
285                Ok(())
286            }
287        }
288    }

290    fn execute_insert(
291        &self,
292        table: &str,
293        columns: Option<Vec<String>>,
294        values: Vec<Value>,
295    ) -> Result<()> {
296        // Check if we're in an explicit transaction; if not, auto-commit
297        let explicit_txn = self.active_transaction.read().clone();
298        let txn: Arc<Transaction>;
299        let auto_commit: bool;
300
301        if let Some(ref active) = explicit_txn {
302            txn = Arc::clone(active);
303            auto_commit = false;
304        } else {
305            txn = self.transaction_manager.begin();
306            auto_commit = true;
307        }
308
309        if !auto_commit {
310            self.lock_manager
311                .acquire_lock(table, txn.id(), LockType::Exclusive)?;
312        } else {
313            self.lock_manager
314                .acquire_lock(table, txn.id(), LockType::Exclusive)?;
315        }
316
317        // Get table schema and immediately clone to release lock
318        let table_schema = {
319            let schema = self.schema.read();
320            schema.get_table(table)?.clone()
321        }; // schema lock released here
322
323        // Validate columns and values
324        let column_names = if let Some(ref cols) = columns {
325            cols.clone()
326        } else {
327            table_schema
328                .columns
329                .iter()
330                .map(|c| c.name.clone())
331                .collect()
332        };
333
334        if column_names.len() != values.len() {
335            self.lock_manager.release_lock(table, txn.id())?;
336            return Err(VelociError::ConstraintViolation(
337                "Column count doesn't match value count".to_string(),
338            ));
339        }
340
341        // Find primary key
342        let pk_index = table_schema
343            .columns
344            .iter()
345            .position(|c| c.primary_key)
346            .ok_or_else(|| VelociError::ConstraintViolation("No primary key defined".to_string()))?;
347
348        let pk_col_name = &table_schema.columns[pk_index].name;
349        let pk_value_index = column_names
350            .iter()
351            .position(|c| c == pk_col_name)
352            .ok_or_else(|| {
353                VelociError::ConstraintViolation("Primary key value not provided".to_string())
354            })?;
355
356        let pk_value = values[pk_value_index].as_integer()?;
357
358        // Validate NOT NULL and vector-dimension constraints before acquiring
359        // the BTree lock.
360        for (i, value) in values.iter().enumerate() {
361            let col_name = &column_names[i];
362            if let Some(col) = table_schema.columns.iter().find(|c| &c.name == col_name) {
363                if col.not_null && matches!(value, Value::Null) {
364                    self.lock_manager.release_lock(table, txn.id())?;
365                    return Err(VelociError::ConstraintViolation(format!(
366                        "Column '{}' cannot be NULL", col_name
367                    )));
368                }
369                if let DataType::Vector(dim) = col.data_type {
370                    match value {
371                        Value::Null => {}
372                        Value::Vector(v) if v.len() == dim as usize => {}
373                        Value::Vector(v) => {
374                            self.lock_manager.release_lock(table, txn.id())?;
375                            return Err(VelociError::ConstraintViolation(format!(
376                                "Column '{}' expects a vector of dimension {}, got {}",
377                                col_name, dim, v.len()
378                            )));
379                        }
380                        other => {
381                            self.lock_manager.release_lock(table, txn.id())?;
382                            return Err(VelociError::TypeMismatch {
383                                expected: format!("Vector({})", dim),
384                                actual: format!("{:?}", other),
385                            });
386                        }
387                    }
388                }
389            }
390        }
391
392        // Create row data
393        let mut row_values = vec![Value::Null; table_schema.columns.len()];
394        for (i, col_name) in column_names.iter().enumerate() {
395            if let Some(col_index) = table_schema.columns.iter().position(|c| &c.name == col_name) {
396                row_values[col_index] = values[i].clone();
397            }
398        }
399        let row = Row::new(row_values);
400
401        // Single acquisition of btrees collection and btree with write intent
402        // No lock upgrade, no multiple acquisitions
403        let result = {
404            let btrees = self.btrees.read();
405            let btree_arc = btrees.get(table).ok_or_else(|| {
406                VelociError::NotFound(format!("Table '{}' not initialized", table))
407            })?;
408
409            // Acquire write lock directly (no upgrade from read)
410            let mut btree = btree_arc.write();
411
412            // Check for primary key uniqueness
413            if btree.search(pk_value)?.is_some() {
414                Err(VelociError::ConstraintViolation(format!(
415                    "Primary key {} already exists in table '{}'",
416                    pk_value, table
417                )))
418            } else {
419                // Insert into B-Tree for persistence
420                btree.insert(pk_value, &row)?;
421                Ok(())
422            }
423        }; // btrees and btree locks released here
424
425        // Handle result
426        if let Err(e) = result {
427            self.lock_manager.release_lock(table, txn.id())?;
428            self.transaction_manager.abort(&txn)?;
429            return Err(e);
430        }
431
432        self.cdc
433            .record(table, ChangeOp::Insert, pk_value, None, Some(row));
434
435        // Only commit/release lock for auto-commit mode
436        if auto_commit {
437            self.transaction_manager.commit(&txn)?;
438            self.lock_manager.release_lock(table, txn.id())?;
439        }
440
441        Ok(())
442    }

444    fn execute_select(
445        &self,
446        table: &str,
447        columns: Vec<String>,
448        where_clause: Option<WhereClause>,
449        order_by: Option<OrderBy>,
450        limit: Option<u64>,
451    ) -> Result<QueryResult> {
452        let explicit_txn = self.active_transaction.read().clone();
453        let txn: Arc<Transaction>;
454        let auto_commit: bool;
455
456        if let Some(ref active) = explicit_txn {
457            txn = Arc::clone(active);
458            auto_commit = false;
459        } else {
460            txn = self.transaction_manager.begin();
461            auto_commit = true;
462        }
463
464        self.lock_manager
465            .acquire_lock(table, txn.id(), LockType::Shared)?;
466
467        // Get table schema and release lock immediately
468        let table_schema = {
469            let schema = self.schema.read();
470            schema.get_table(table)?.clone()
471        }; // schema lock released
472
473        // The B-tree is authoritative. MVCC overlay is intentionally not used
474        // until WAL-backed snapshot isolation lands.
475        let all_rows: Vec<(i64, Row)> = {
476            let btrees = self.btrees.read();
477            let btree_arc = btrees.get(table).ok_or_else(|| {
478                VelociError::NotFound(format!("Table '{}' not initialized", table))
479            })?;
480            let btree = btree_arc.read();
481            btree.scan()?
482        };
483
484        // Process data without holding any locks. Filtering runs in parallel
485        // (rayon) once the row count crosses PARALLEL_THRESHOLD.
486        let mut filtered_rows: Vec<(i64, Row)> = if let Some(ref where_clause) = where_clause {
487            if all_rows.len() >= PARALLEL_THRESHOLD {
488                all_rows
489                    .into_par_iter()
490                    .filter(|(_, row)| {
491                        self.evaluate_where_clause(row, where_clause, &table_schema)
492                            .unwrap_or(false)
493                    })
494                    .collect()
495            } else {
496                all_rows
497                    .into_iter()
498                    .filter(|(_, row)| {
499                        self.evaluate_where_clause(row, where_clause, &table_schema)
500                            .unwrap_or(false)
501                    })
502                    .collect()
503            }
504        } else {
505            all_rows
506        };
507
508        // Apply ORDER BY. Two forms are supported:
509        // 1. A vector distance expression -> exact KNN (parallel distances).
510        // 2. A plain column -> comparison sort (parallel for large sets).
511        if let Some(ref order) = order_by {
512            if let Some(parsed) = vector::parse_distance_expr(&order.column) {
513                let expr = parsed?;
514                let col_index = table_schema
515                    .columns
516                    .iter()
517                    .position(|c| c.name == expr.column)
518                    .ok_or_else(|| {
519                        VelociError::NotFound(format!(
520                            "Vector column '{}' not found in table '{}'",
521                            expr.column, table
522                        ))
523                    })?;
524
525                if order.ascending {
526                    // Nearest-first with a LIMIT is the classic KNN shape:
527                    // use top-k selection instead of a full sort.
528                    let k = limit.map(|n| n as usize).unwrap_or(usize::MAX).min(filtered_rows.len());
529                    filtered_rows = vector::knn(
530                        filtered_rows,
531                        col_index,
532                        &expr.query,
533                        expr.metric,
534                        k,
535                    )
536                    .into_iter()
537                    .map(|(_, key, row)| (key, row))
538                    .collect();
539                } else {
540                    let distances = vector::compute_distances(
541                        &filtered_rows,
542                        col_index,
543                        &expr.query,
544                        expr.metric,
545                    );
546                    let mut scored: Vec<(f64, i64, Row)> = filtered_rows
547                        .into_iter()
548                        .zip(distances)
549                        .map(|((key, row), d)| (d, key, row))
550                        .collect();
551                    scored.sort_by(|a, b| {
552                        b.0.partial_cmp(&a.0)
553                            .unwrap_or(std::cmp::Ordering::Equal)
554                            .then(a.1.cmp(&b.1))
555                    });
556                    filtered_rows = scored.into_iter().map(|(_, key, row)| (key, row)).collect();
557                }
558            } else {
559                let col_index = table_schema
560                    .columns
561                    .iter()
562                    .position(|c| c.name == order.column)
563                    .ok_or_else(|| {
564                        VelociError::NotFound(format!(
565                            "ORDER BY column '{}' not found in table '{}'",
566                            order.column, table
567                        ))
568                    })?;
569
570                let cmp = |(ak, a): &(i64, Row), (bk, b): &(i64, Row)| {
571                    let av = a.values.get(col_index).unwrap_or(&Value::Null);
572                    let bv = b.values.get(col_index).unwrap_or(&Value::Null);
573                    let primary = compare_values(av, bv);
574                    let secondary = ak.cmp(bk);
575                    let combined = primary.then(secondary);
576                    if order.ascending { combined } else { combined.reverse() }
577                };
578
579                if filtered_rows.len() >= PARALLEL_THRESHOLD {
580                    filtered_rows.par_sort_by(cmp);
581                } else {
582                    filtered_rows.sort_by(cmp);
583                }
584            }
585        }
586
587        // Apply LIMIT n (does not affect COUNT(*), which counts rows after WHERE).
588        if let Some(n) = limit {
589            filtered_rows.truncate(n as usize);
590        }
591
592        // Check for aggregate functions (COUNT)
593        let is_count = columns.len() == 1
594            && columns[0].to_uppercase().starts_with("COUNT(");
595
596        if is_count {
597            let count = filtered_rows.len() as i64;
598            let result_columns = vec![Column {
599                name: columns[0].clone(),
600                data_type: crate::types::DataType::Integer,
601                primary_key: false,
602                not_null: true,
603                unique: false,
604            }];
605            let result_rows = vec![Row::new(vec![Value::Integer(count)])];
606
607            if auto_commit {
608                self.transaction_manager.commit(&txn)?;
609                self.lock_manager.release_lock(table, txn.id())?;
610            }
611
612            return Ok(QueryResult::new(result_columns, result_rows));
613        }
614
615        // Project columns. Each projection item is either '*', a table
616        // column, or a vector distance expression computed per row.
617        enum ProjItem {
618            Star,
619            Col(usize),
620            Distance {
621                col_index: usize,
622                metric: DistanceMetric,
623                query: Vec<f32>,
624            },
625        }
626
627        let mut proj_items: Vec<(String, ProjItem)> = Vec::with_capacity(columns.len());
628        for col_name in &columns {
629            if col_name == "*" {
630                proj_items.push(("*".to_string(), ProjItem::Star));
631            } else if let Some(idx) = table_schema.columns.iter().position(|c| &c.name == col_name)
632            {
633                proj_items.push((col_name.clone(), ProjItem::Col(idx)));
634            } else if let Some(parsed) = vector::parse_distance_expr(col_name) {
635                let expr = parsed?;
636                let col_index = table_schema
637                    .columns
638                    .iter()
639                    .position(|c| c.name == expr.column)
640                    .ok_or_else(|| {
641                        VelociError::NotFound(format!(
642                            "Vector column '{}' not found in table '{}'",
643                            expr.column, table
644                        ))
645                    })?;
646                proj_items.push((
647                    col_name.clone(),
648                    ProjItem::Distance {
649                        col_index,
650                        metric: expr.metric,
651                        query: expr.query,
652                    },
653                ));
654            } else {
655                return Err(VelociError::NotFound(format!(
656                    "Column '{}' not found in table '{}'",
657                    col_name, table
658                )));
659            }
660        }
661
662        let mut result_columns: Vec<Column> = Vec::new();
663        for (name, item) in &proj_items {
664            match item {
665                ProjItem::Star => result_columns.extend(table_schema.columns.iter().cloned()),
666                ProjItem::Col(idx) => result_columns.push(table_schema.columns[*idx].clone()),
667                ProjItem::Distance { .. } => result_columns.push(Column {
668                    name: name.clone(),
669                    data_type: DataType::Real,
670                    primary_key: false,
671                    not_null: false,
672                    unique: false,
673                }),
674            }
675        }
676
677        let result_rows: Vec<Row> = filtered_rows
678            .into_iter()
679            .map(|(_, row)| {
680                let mut projected: Vec<Value> = Vec::with_capacity(result_columns.len());
681                for (_, item) in &proj_items {
682                    match item {
683                        ProjItem::Star => projected.extend(row.values.iter().cloned()),
684                        ProjItem::Col(idx) => {
685                            projected.push(row.values.get(*idx).cloned().unwrap_or(Value::Null))
686                        }
687                        ProjItem::Distance {
688                            col_index,
689                            metric,
690                            query,
691                        } => {
692                            let d = row
693                                .values
694                                .get(*col_index)
695                                .and_then(|v| vector::value_as_vector(v).ok())
696                                .and_then(|v| metric.distance(&v, query).ok());
697                            projected.push(match d {
698                                Some(d) => Value::Float(d),
699                                None => Value::Null,
700                            });
701                        }
702                    }
703                }
704                Row::new(projected)
705            })
706            .collect();
707
708        if auto_commit {
709            self.transaction_manager.commit(&txn)?;
710            self.lock_manager.release_lock(table, txn.id())?;
711        }
712
713        Ok(QueryResult::new(result_columns, result_rows))
714    }

716    fn execute_update(
717        &self,
718        table: &str,
719        assignments: HashMap<String, Value>,
720        where_clause: Option<WhereClause>,
721    ) -> Result<()> {
722        let explicit_txn = self.active_transaction.read().clone();
723        let txn: Arc<Transaction>;
724        let auto_commit: bool;
725        if let Some(ref active) = explicit_txn {
726            txn = Arc::clone(active);
727            auto_commit = false;
728        } else {
729            txn = self.transaction_manager.begin();
730            auto_commit = true;
731        }
732        self.lock_manager
733            .acquire_lock(table, txn.id(), LockType::Exclusive)?;
734
735        // Get table schema and release lock
736        let table_schema = {
737            let schema = self.schema.read();
738            schema.get_table(table)?.clone()
739        }; // schema lock released
740
741        // Change events collected during the update, recorded to CDC only
742        // after the whole statement succeeds.
743        let mut cdc_events: Vec<(i64, Row, Row)> = Vec::new();
744
745        // Perform update with single btree lock acquisition
746        let result = {
747            let btrees = self.btrees.read();
748            let btree_arc = btrees.get(table).ok_or_else(|| {
749                VelociError::NotFound(format!("Table '{}' not initialized", table))
750            })?;
751            let mut btree = btree_arc.write();
752
753            // Scan all rows
754            let all_rows = btree.scan()?;
755
756            // Find rows to update
757            let rows_to_update: Vec<(i64, Row)> = if let Some(ref where_clause) = where_clause {
758                all_rows
759                    .into_iter()
760                    .filter(|(_, row)| self.evaluate_where_clause(row, where_clause, &table_schema).unwrap_or(false))
761                    .collect()
762            } else {
763                all_rows
764            };
765
766            // Find primary key column
767            let pk_index = table_schema
768                .columns
769                .iter()
770                .position(|c| c.primary_key)
771                .ok_or_else(|| VelociError::ConstraintViolation("No primary key defined".to_string()))?;
772
773            // Update each row
774            for (key, row) in &rows_to_update {
775                let mut updated_row = row.clone();
776                let mut new_pk_value = *key; // Default to existing key
777                let mut pk_being_updated = false;
778
779                // Apply updates to the row
780                for (col_name, new_value) in &assignments {
781                    if let Some(col_index) = table_schema.columns.iter().position(|c| &c.name == col_name) {
782                        // Check NOT NULL constraint
783                        let col = &table_schema.columns[col_index];
784                        if col.not_null && matches!(new_value, Value::Null) {
785                            return Err(VelociError::ConstraintViolation(format!(
786                                "Column '{}' cannot be NULL", col_name
787                            )));
788                        }
789
790                        updated_row.values[col_index] = new_value.clone();
791
792                        // Check if primary key is being updated
793                        if col_index == pk_index {
794                            new_pk_value = new_value.as_integer()?;
795                            pk_being_updated = true;
796                        }
797                    }
798                }
799
800                // If primary key is being updated, check for uniqueness
801                if pk_being_updated && new_pk_value != *key {
802                    if btree.search(new_pk_value)?.is_some() {
803                        return Err(VelociError::ConstraintViolation(format!(
804                            "Primary key {} already exists in table '{}'",
805                            new_pk_value, table
806                        )));
807                    }
808                }
809
810                // Delete old row and insert updated row
811                btree.delete(*key)?;
812                btree.insert(new_pk_value, &updated_row)?;
813
814                if self.cdc.is_enabled() {
815                    cdc_events.push((*key, row.clone(), updated_row));
816                }
817            }
818            
819            Ok::<(), VelociError>(())
820        }; // btrees and btree locks released
821
822        // Handle errors
823        if let Err(e) = result {
824            self.lock_manager.release_lock(table, txn.id())?;
825            self.transaction_manager.abort(&txn)?;
826            return Err(e);
827        }
828
829        for (key, before, after) in cdc_events {
830            self.cdc
831                .record(table, ChangeOp::Update, key, Some(before), Some(after));
832        }
833
834        if auto_commit {
835            self.transaction_manager.commit(&txn)?;
836            self.lock_manager.release_lock(table, txn.id())?;
837        }
838
839        Ok(())
840    }

842    #[allow(clippy::let_and_return)]
843    fn execute_delete(&self, table: &str, where_clause: Option<WhereClause>) -> Result<()> {
844        let explicit_txn = self.active_transaction.read().clone();
845        let txn: Arc<Transaction>;
846        let auto_commit: bool;
847        if let Some(ref active) = explicit_txn {
848            txn = Arc::clone(active);
849            auto_commit = false;
850        } else {
851            txn = self.transaction_manager.begin();
852            auto_commit = true;
853        }
854        self.lock_manager
855            .acquire_lock(table, txn.id(), LockType::Exclusive)?;
856
857        // Get table schema and release lock
858        let table_schema = {
859            let schema = self.schema.read();
860            schema.get_table(table)?.clone()
861        }; // schema lock released
862
863        // Perform delete with single btree lock acquisition
864        let mut cdc_events: Vec<(i64, Row)> = Vec::new();
865        let result: Result<()> = {
866            let btrees = self.btrees.read();
867            let btree_arc = btrees.get(table).ok_or_else(|| {
868                VelociError::NotFound(format!("Table '{}' not initialized", table))
869            })?;
870            let mut btree = btree_arc.write();
871
872            let all_rows = btree.scan()?;
873            let rows_to_delete: Vec<(i64, Row)> = if let Some(ref where_clause) = where_clause {
874                all_rows
875                    .into_iter()
876                    .filter(|(_, row)| self.evaluate_where_clause(row, where_clause, &table_schema).unwrap_or(false))
877                    .collect()
878            } else {
879                all_rows
880            };
881
882            for (key, row) in rows_to_delete {
883                btree.delete(key)?;
884                if self.cdc.is_enabled() {
885                    cdc_events.push((key, row));
886                }
887            }
888            Ok(())
889        };
890
891        if let Err(e) = result {
892            self.lock_manager.release_lock(table, txn.id())?;
893            self.transaction_manager.abort(&txn)?;
894            return Err(e);
895        }
896
897        for (key, before) in cdc_events {
898            self.cdc
899                .record(table, ChangeOp::Delete, key, Some(before), None);
900        }
901
902        if auto_commit {
903            self.transaction_manager.commit(&txn)?;
904            self.lock_manager.release_lock(table, txn.id())?;
905        }
906
907        Ok(())
908    }

910    fn evaluate_where_clause(
911        &self,
912        row: &Row,
913        where_clause: &WhereClause,
914        table_schema: &TableSchema,
915    ) -> Result<bool> {
916        // (helper) See `compare_values` below for sort ordering semantics.
917        for condition in &where_clause.conditions {
918            let col_index = table_schema
919                .columns
920                .iter()
921                .position(|c| c.name == condition.column)
922                .ok_or_else(|| {
923                    VelociError::NotFound(format!("Column '{}' not found", condition.column))
924                })?;
925
926            let row_value = &row.values[col_index];
927            let condition_value = &condition.value;
928
929            if !condition.operator.evaluate(row_value, condition_value)? {
930                return Ok(false);
931            }
932        }
933
934        Ok(true)
935    }
936}

938/// Order two `Value`s for ORDER BY purposes.
939///
940/// SQL NULL semantics here: NULL sorts after all non-NULL values (the SQLite
941/// default for ASC and the conventional "NULLS LAST" behaviour).
942fn compare_values(a: &Value, b: &Value) -> std::cmp::Ordering {
943    use std::cmp::Ordering;
944    match (a, b) {
945        (Value::Null, Value::Null) => Ordering::Equal,
946        (Value::Null, _) => Ordering::Greater,
947        (_, Value::Null) => Ordering::Less,
948        (Value::Integer(x), Value::Integer(y)) => x.cmp(y),
949        (Value::Text(x), Value::Text(y)) => x.cmp(y),
950        (Value::Blob(x), Value::Blob(y)) => x.cmp(y),
951        // Mixed numeric: promote to f64.
952        (lhs, rhs)
953            if matches!(lhs, Value::Float(_) | Value::Real(_) | Value::Integer(_))
954                && matches!(rhs, Value::Float(_) | Value::Real(_) | Value::Integer(_)) =>
955        {
956            let lf = lhs.as_float().unwrap_or(0.0);
957            let rf = rhs.as_float().unwrap_or(0.0);
958            lf.partial_cmp(&rf).unwrap_or(Ordering::Equal)
959        }
960        // Different incomparable types: fall back to discriminant order so that
961        // the sort is at least total and deterministic.
962        _ => format!("{:?}", a).cmp(&format!("{:?}", b)),
963    }
964}
